create or replace function _rep_apply(p_char uuid, p_hero int, p_villain int, p_heat int) returns void
language plpgsql security definer set search_path = public as $$
declare o uuid;
begin
  select owner into o from characters where id = p_char; if o is null then return; end if;
  insert into reputation (char_id, owner) values (p_char, o) on conflict (char_id) do nothing;
  update reputation set
    hero = greatest(0, hero + coalesce(p_hero, 0)),
    villain = greatest(0, villain + coalesce(p_villain, 0)),
    heat = greatest(0, _heat_eff(heat, heat_at) + coalesce(p_heat, 0)),
    heat_at = now(), updated_at = now()
  where char_id = p_char;
end $$;

create or replace function _mai_ranks() returns jsonb language sql immutable as $$
  select '{"Pacífica":{"xp":20,"tj":500,"lv":1},"Fácil":{"xp":70,"tj":2500,"lv":1},"Média":{"xp":120,"tj":6000,"lv":2},"Arriscada":{"xp":130,"tj":8000,"lv":3},"Difícil":{"xp":200,"tj":20000,"lv":4}}'::jsonb $$;

create or replace function _mai_pick(arr jsonb) returns jsonb language sql volatile as
$$ select arr -> (floor(random() * jsonb_array_length(arr)))::int $$;

create or replace function _mai_render(t text, v jsonb) returns text language plpgsql immutable as $$
declare k text; val text;
begin
  if t is null then return ''; end if;
  for k, val in select key, value from jsonb_each_text(v) loop t := replace(t, '{' || k || '}', val); end loop;
  return t;
end $$;

create or replace function _mai_bonus(p_stats jsonb, p_stat text) returns int language sql immutable as
$$ select least(6, floor(coalesce((p_stats ->> p_stat)::numeric, 0) / 20))::int $$;

create or replace function _mai_hint(p_dc int, p_bonus int) returns text language sql immutable as $$
  select case when p_dc is null then 'Sem teste'
              when 21 - (p_dc - p_bonus) >= 15 then 'Favorável'
              when 21 - (p_dc - p_bonus) >= 10 then 'Arriscado'
              when 21 - (p_dc - p_bonus) >= 5  then 'Difícil'
              else 'Quase impossível' end $$;

-- o que o jogador enxerga: texto + opções (nunca as saídas nem a dificuldade exata)
create or replace function _mai_view(s mission_ai_sessions) returns jsonb language plpgsql security definer set search_path = public as $$
declare plan jsonb; ch characters; node jsonb; opts jsonb := '[]'::jsonb; i int; o jsonb;
begin
  select data into plan from mission_ai_plans where session_id = s.id;
  select * into ch from characters where id = s.char_id;
  node := plan -> 'nodes' -> s.node;
  if s.status = 'done' then
    return jsonb_build_object('session', s.id, 'status', 'done', 'title', s.title, 'rank', s.rank_name, 'hp', s.hp, 'made', jsonb_array_length(s.log),
      'total', (plan ->> 'total')::int, 'result', s.result, 'reward', s.reward, 'text', _mai_render(node ->> 'text', s.vars), 'log', s.log);
  end if;
  for i in 0 .. coalesce(jsonb_array_length(node -> 'options'), 0) - 1 loop
    o := node -> 'options' -> i;
    opts := opts || jsonb_build_array(jsonb_build_object('i', i, 'text', _mai_render(o ->> 't', s.vars), 'stat', o ->> 'stat',
      'hint', _mai_hint((o ->> 'dc')::int, _mai_bonus(ch.stats, o ->> 'stat'))));
  end loop;
  return jsonb_build_object('session', s.id, 'status', 'active', 'title', s.title, 'rank', s.rank_name, 'hp', s.hp, 'made', jsonb_array_length(s.log),
    'total', (plan ->> 'total')::int, 'text', _mai_render(node ->> 'text', s.vars), 'options', opts, 'log', s.log);
end $$;

create or replace function mission_ai_list(p_char uuid) returns jsonb language plpgsql security definer set search_path = public as $$
declare cfg mission_ai_config; used int; act uuid; rk jsonb := '[]'::jsonb; k text; r jsonb; ch characters;
begin
  if auth.uid() is null then raise exception 'Não autenticado'; end if;
  select * into ch from characters where id = p_char and owner = auth.uid(); if not found then raise exception 'Personagem inválido'; end if;
  select * into cfg from mission_ai_config where id = 1;
  select count(*) into used from mission_ai_sessions where char_id = p_char and started_at > now() - interval '24 hours';
  select id into act from mission_ai_sessions where char_id = p_char and status = 'active' limit 1;
  for k, r in select key, value from jsonb_each(_mai_ranks()) loop
    rk := rk || jsonb_build_array(jsonb_build_object('rank', k, 'min_level', (r ->> 'lv')::int,
      'xp', round((r ->> 'xp')::numeric * cfg.reward_pct / 100.0), 'tarjas', round((r ->> 'tj')::numeric * cfg.reward_pct / 100.0), 'locked', coalesce(ch.level, 1) < (r ->> 'lv')::int));
  end loop;
  return jsonb_build_object('enabled', cfg.enabled, 'pct', cfg.reward_pct, 'limit', cfg.daily_limit, 'used', used, 'active', act, 'ranks', rk);
end $$;

create or replace function mission_ai_start(p_char uuid, p_rank text) returns jsonb language plpgsql security definer set search_path = public as $$
declare ch characters; cfg mission_ai_config; s mission_ai_sessions; used int; rk jsonb; plan jsonb; v jsonb;
begin
  if auth.uid() is null then raise exception 'Não autenticado'; end if;
  select * into ch from characters where id = p_char and owner = auth.uid(); if not found then raise exception 'Personagem inválido'; end if;
  select * into cfg from mission_ai_config where id = 1;
  if not cfg.enabled then raise exception 'O Modo Automático está desativado pelo administrador.'; end if;
  select * into s from mission_ai_sessions where char_id = p_char and status = 'active' limit 1;
  if found then return _mai_view(s); end if;                       -- retoma a missão em andamento
  rk := _mai_ranks() -> p_rank; if rk is null then raise exception 'Dificuldade inválida.'; end if;
  if coalesce(ch.level, 1) < (rk ->> 'lv')::int then raise exception 'Essa dificuldade exige nível %.', (rk ->> 'lv')::int; end if;
  select count(*) into used from mission_ai_sessions where char_id = p_char and started_at > now() - interval '24 hours';
  if used >= cfg.daily_limit then raise exception 'Limite diário de missões automáticas atingido (%).', cfg.daily_limit; end if;
  plan := _mai_generate(p_rank);
  v := (plan -> 'vars') || jsonb_build_object('player', ch.name, 'clan_line', coalesce(plan -> 'clan_lines' ->> ch.clan, plan -> 'clan_lines' ->> 'default', ''));
  plan := plan - 'clan_lines';
  insert into mission_ai_sessions (owner, char_id, rank_name, title, vars, node, hp, base_xp, base_tarjas)
    values (auth.uid(), p_char, p_rank, plan ->> 'title', v, 'n1', (plan ->> 'hp')::int, (rk ->> 'xp')::int, (rk ->> 'tj')::int) returning * into s;
  insert into mission_ai_plans (session_id, data) values (s.id, plan);
  return _mai_view(s);
end $$;

create or replace function mission_ai_view(p_session uuid) returns jsonb language plpgsql security definer set search_path = public as $$
declare s mission_ai_sessions;
begin
  if auth.uid() is null then raise exception 'Não autenticado'; end if;
  select * into s from mission_ai_sessions where id = p_session and (owner = auth.uid() or is_staff());
  if not found then raise exception 'Missão não encontrada'; end if;
  return _mai_view(s);
end $$;

    v_hero := coalesce((nn ->> 'hero')::int, 0) + s.rep_h; v_vil := coalesce((nn ->> 'villain')::int, 0) + s.rep_v;

      insert into intel_clues (subject_name, subject_type, subject_char, action_cat, action_label, object_name, item_name, place, visibility, tier, price, excerpt, happened_at, created_by)
      values (ch.name, 'player', ch.id, nn -> 'trace' ->> 'cat', nn -> 'trace' ->> 'label',
              nullif(s.vars ->> (nn -> 'trace' ->> 'obj'), ''), nullif(s.vars ->> (nn -> 'trace' ->> 'item'), ''), s.vars ->> 'place', vis, tier,
              case tier when 1 then 300 when 2 then 1500 else 6000 end, left(_mai_render(nn ->> 'text', s.vars), 260), now(), auth.uid());
    end if;
    rw := jsonb_build_object('xp', v_xp, 'tarjas', v_tj, 'hero', v_hero, 'villain', v_vil, 'heat', v_heat, 'visibility', vis);
  end if;

  update mission_ai_sessions set node = s.node, hp = s.hp, exposure = s.exposure, score = s.score, rep_h = s.rep_h, rep_v = s.rep_v, rep_t = s.rep_t, log = s.log,
    status = case when ending then 'done' else 'active' end, result = case when ending then kind else null end,
    reward = rw, finished_at = case when ending then now() else null end
  where id = s.id returning * into s;
  return jsonb_build_object('step', stp, 'view', _mai_view(s));
end $$;

create or replace function mission_ai_abandon(p_session uuid) returns void language plpgsql security definer set search_path = public as $$
begin
  if auth.uid() is null then raise exception 'Não autenticado'; end if;
  update mission_ai_sessions set status = 'done', result = 'abandon', finished_at = now(),
    reward = jsonb_build_object('xp', 0, 'tarjas', 0, 'hero', 0, 'villain', 0, 'heat', 0, 'visibility', 'discreto')
  where id = p_session and owner = auth.uid() and status = 'active';
end $$;

create or replace function admin_mai_config(p_enabled boolean, p_pct int, p_limit int) returns void language plpgsql security definer set search_path = public as $$
begin
  if not is_admin() then raise exception 'Apenas administradores.'; end if;
  update mission_ai_config set enabled = p_enabled, reward_pct = greatest(0, least(100, p_pct)), daily_limit = greatest(0, least(20, p_limit)) where id = 1;
end $$;
